const report: Record<string, unknown> = {};
const emitReport = () => {
  console.log("DASHBOARD_REFRESH_AUDIT", JSON.stringify(report));
  if (process.env.AUDIT_REPORT_PATH) writeFileSync(process.env.AUDIT_REPORT_PATH, JSON.stringify(report, null, 2) + "\n");
};
const worktree = (path: string) => ({ path, head: "abc123", shortHead: "abc123", branch: "main", detached: false, locked: false, prunable: false, dirty: false });
function clients(events: EventStream, count: number, fetch: () => void) {
  for (let i = 0; i < count; i += 1) {
    const client = new EventEmitter();
    Object.assign(client, { write: (message: string) => { if (message.startsWith("event: changed")) fetch(); return true; }, end: () => client.emit("close") });
    events.add(client as unknown as ServerResponse);
  }
}

it("measures overlapping dashboard scans and the failed-storage feedback loop", async () => {
  const root = mkdtempSync(join(tmpdir(), "switcher-refresh-audit-"));
  const store = new SqliteStateStore(join(root, "state.sqlite3"));
  const events = new EventStream();
  const processes = new ProcessManager();
  let storage: WorktreeStorageManager | undefined;
  vi.useFakeTimers();
  try {
    for (let i = 0; i < 2; i += 1) store.addProject({ name: `P${i}`, repositoryPath: `/fixture/${i}`, port: 3300 + i, executable: "node", args: [] });
    let active = 0, peak = 0;
    const git = { canonicalRepositoryPath: async (p: string) => p, list: vi.fn(async (path: string) => {
      active += 1; peak = Math.max(peak, active);
      await new Promise((resolve) => setTimeout(resolve, 650));
      active -= 1;
      return [0, 1, 2].map((i) => worktree(`${path}/${i}`));
    }) };
    const presets = new ProjectTestCommandResolver();
    const discover = vi.spyOn(presets, "discover").mockReturnValue([]);
    const service = new ControlService(store, git, processes, undefined, undefined, undefined, undefined, presets);
    const requests: Array<Promise<unknown>> = [];
    clients(events, 3, () => { requests.push(service.dashboard()); });
    for (let tick = 0; tick < 4; tick += 1) { events.publish(); await vi.advanceTimersByTimeAsync(250); }
    events.close();
    await vi.advanceTimersByTimeAsync(650);
    await Promise.all(requests);
    report.overlap = { clients: 3, projects: 2, worktreesPerProject: 3, changedEvents: 4, readerDelayMs: 650, dashboardRequests: requests.length, repositoryReads: git.list.mock.calls.length, peakConcurrentRepositoryReads: peak, presetDiscoveries: discover.mock.calls.length };
    expect(requests).toHaveLength(12); expect(git.list).toHaveBeenCalledTimes(24); expect(peak).toBe(18); expect(discover).toHaveBeenCalledTimes(72);

    const loopEvents = new EventStream();
    const scan = vi.fn(async () => { throw new Error("fixture unavailable"); });
    storage = new WorktreeStorageManager(store, { scan }, loopEvents.publish);
    const fastGit = { canonicalRepositoryPath: async (p: string) => p, list: vi.fn(async (path: string) => [worktree(path)]) };
    const loopService = new ControlService(store, fastGit, processes, undefined, undefined, storage, undefined, presets);
    const loopRequests: Array<Promise<unknown>> = [];
    clients(loopEvents, 1, () => { loopRequests.push(loopService.dashboard()); });
    await loopService.dashboard(); // The only initial stimulus; no log events.
    await vi.advanceTimersByTimeAsync(1000);
    loopEvents.close();
    await Promise.all(loopRequests);
    await storage.close();
    report.storageFailure = { projects: 2, elapsedVirtualMs: 1000, externalLogEvents: 0, followupDashboardRequests: loopRequests.length, repositoryReads: fastGit.list.mock.calls.length, failedStorageScans: scan.mock.calls.length };
    expect(loopRequests.length).toBeGreaterThanOrEqual(3);
    expect(scan.mock.calls.length).toBeGreaterThanOrEqual(8);
    emitReport();
  } finally {
    await storage?.close(); events.close(); await processes.stopAll(); store.close(); vi.useRealTimers(); rmSync(root, { recursive: true, force: true });
  }
});

it("counts real Git processes and service response bytes for zero, one, and three clients", async () => {
  const root = mkdtempSync(join(tmpdir(), "switcher-refresh-git-"));
  const store = new SqliteStateStore(join(root, "state.sqlite3"));
  const processes = new ProcessManager();
  const git = new SystemGitWorktreeReader();
  const calls = vi.spyOn(git, "list");
  const presets = new ProjectTestCommandResolver();
  const discoveries = vi.spyOn(presets, "discover");
  const service = new ControlService(store, git, processes, undefined, undefined, undefined, undefined, presets);
  const scenarios: unknown[] = [];
  try {
    for (let index = 0; index < 2; index += 1) {
      const repo = join(root, `repo-${index}`); mkdirSync(repo);
      execFileSync("git", ["init", "-q", repo]);
      writeFileSync(join(repo, "package.json"), JSON.stringify({ scripts: { test: "node test.js", build: "node build.js" } }));
      for (let i = 0; i < 100; i += 1) writeFileSync(join(repo, `source-${i}.txt`), "x".repeat(1024));
      execFileSync("git", ["-C", repo, "add", "."]);
      execFileSync("git", ["-C", repo, "-c", "user.name=Audit Fixture", "-c", "user.email=audit@example.invalid", "-c", "commit.gpgsign=false", "commit", "-qm", "fixture"]);
      for (let i = 1; i <= 2; i += 1) execFileSync("git", ["-C", repo, "worktree", "add", "--quiet", "--detach", join(root, `repo-${index}-wt-${i}`)]);
      store.addProject({ name: `P${index}`, repositoryPath: repo, port: 3300 + index, executable: "node", args: [] });
    }
    // Worktree paths have equal lengths within one scenario; no runtime processes,
    // test history, storage scans or network listeners are included in this probe.
    for (const count of [0, 1, 3]) {
      const trace = join(root, `trace-${count}.jsonl`);
      vi.stubEnv("GIT_TRACE2_EVENT", trace);
      calls.mockClear(); discoveries.mockClear();
      const events = new EventStream();
      let bytes = 0;
      const requests: Array<Promise<unknown>> = [];
      clients(events, count, () => { requests.push(service.dashboard().then((body) => { bytes += Buffer.byteLength(JSON.stringify(body)); })); });
      const cpu = process.cpuUsage(); const start = performance.now(); const before = process.memoryUsage().rss;
      let peak = before;
      const sample = setInterval(() => { peak = Math.max(peak, process.memoryUsage().rss); }, 10);
      try {
        for (let tick = 0; tick < 4; tick += 1) { events.publish(); await new Promise((resolve) => setTimeout(resolve, 270)); }
        await Promise.all(requests);
      } finally { clearInterval(sample); events.close(); }
      const usage = process.cpuUsage(cpu);
      const records = count ? readFileSync(trace, "utf8").trim().split("\n").map((line) => JSON.parse(line)) : [];
      const starts = records.filter((record) => record.event === "start");
      const status = starts.filter((record) => record.argv.includes("status")).length;
      const listing = starts.filter((record) => record.argv.includes("worktree")).length;
      scenarios.push({ clients: count, changedEvents: 4, dashboardRequests: requests.length, repositoryReads: calls.mock.calls.length, gitWorktreeCommands: listing, gitStatusCommands: status, presetDiscoveries: discoveries.mock.calls.length, serviceJsonBytes: bytes, wallMs: Math.round(performance.now() - start), nodeCpuUserMs: usage.user / 1000, nodeCpuSystemMs: usage.system / 1000, nodeRssBeforeBytes: before, nodeRssPeakBytes: peak, nodeRssAfterBytes: process.memoryUsage().rss });
      expect(listing).toBe(count * 4 * 2); expect(status).toBe(count * 4 * 2 * 3);
    }
    report.realGit = { projects: 2, worktreesPerProject: 3, trackedFilesPerWorktree: 101, note: "Node CPU/RSS include Vitest; CPU excludes Git children. Four changed events 270ms apart, synthetic clients, no HTTP/browser/runtime/history/storage load. One baseline run, not a production benchmark or an after-fix measurement.", scenarios };
    emitReport();
  } finally { vi.unstubAllEnvs(); await processes.stopAll(); store.close(); rmSync(root, { recursive: true, force: true }); }
}, 15_000);
