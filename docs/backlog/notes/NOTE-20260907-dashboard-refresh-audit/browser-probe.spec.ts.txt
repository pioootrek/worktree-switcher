import { test, expect, type Route } from "@playwright/test";
import { dashboardFixture, mountDashboard } from "./dashboard-fixture";

test("audit reproduces stale dashboard response overwriting a newer result", async ({ page }) => {
  await page.addInitScript(() => {
    const targets = new Set<EventTarget>();
    const original = EventTarget.prototype.addEventListener;
    EventTarget.prototype.addEventListener = function (...args: Parameters<typeof original>) {
      if (args[0] === "changed") targets.add(this);
      return original.apply(this, args);
    };
    Object.assign(window, { auditChanged: () => { for (const target of targets) target.dispatchEvent(new Event("changed")); } });
  });
  const { errors } = await mountDashboard(page);
  await expect(page.getByText("Fixture Web", { exact: true })).toBeVisible();
  const pending: Route[] = [];
  await page.route("**/api/dashboard", (route) => { pending.push(route); });
  await page.evaluate(() => {
    const emit = (window as unknown as { auditChanged: () => void }).auditChanged;
    emit(); emit();
  });
  await expect.poll(() => pending.length).toBe(2);
  const newer = dashboardFixture(); newer.projects[0].project.name = "Newer response";
  await pending[1].fulfill({ json: newer });
  await expect(page.getByText("Newer response", { exact: true })).toBeVisible();
  const older = dashboardFixture(); older.projects[0].project.name = "Older response";
  await pending[0].fulfill({ json: older });
  await expect(page.getByText("Older response", { exact: true })).toBeVisible();
  expect(errors).toEqual([]);
});
